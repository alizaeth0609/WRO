// ================== PROGRAMA 1: Servo con Sensor Ultrasónico Derecha ==================
#include <Servo.h>

#define trigDer A1
#define echoDer A0

Servo miServo;
int servoPin = 9;

void setup() {
  pinMode(trigDer, OUTPUT);
  pinMode(echoDer, INPUT);
  miServo.attach(servoPin);
  Serial.begin(9600);
}

long leerDistancia(int trigPin, int echoPin) {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);
  long duracion = pulseIn(echoPin, HIGH);
  return duracion * 0.034 / 2;
}

void loop() {
  long distancia = leerDistancia(trigDer, echoDer);
  Serial.println(distancia);

  if (distancia < 105) {
    miServo.write(0); // gira a la derecha
  } else {
    miServo.write(90); // queda centrado
  }
  delay(100);
}
