// ================== PROGRAMA 2: Servo con 2 Sensores Ultrasónicos (Derecha/Izquierda) ==================
#include <Servo.h>

#define trigDer 2
#define echoDer 3
#define trigIzq 4
#define echoIzq 5

Servo miServo;
int servoPin = 9;

long leerDistancia(int trigPin, int echoPin) {
  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);
  long duracion = pulseIn(echoPin, HIGH);
  return duracion * 0.034 / 2;
}

void setup() {
  pinMode(trigDer, OUTPUT);
  pinMode(echoDer, INPUT);
  pinMode(trigIzq, OUTPUT);
  pinMode(echoIzq, INPUT);
  miServo.attach(servoPin);
  Serial.begin(9600);
}

void loop() {
  long distanciaDer = leerDistancia(trigDer, echoDer);
  long distanciaIzq = leerDistancia(trigIzq, echoIzq);

  Serial.print("Derecha: "); Serial.print(distanciaDer);
  Serial.print("  Izquierda: "); Serial.println(distanciaIzq);

  if (distanciaDer > distanciaIzq + 5) {
    miServo.write(0); // gira a la derecha
  } else if (distanciaIzq > distanciaDer + 5) {
    miServo.write(180); // gira a la izquierda
  } else {
    miServo.write(90); // queda centrado
  }
  delay(100);
}
